*Step 7 = (Download MySQL JDBC Driver):-
                                        Download the MySQL Connector/J archive =   wget https://dev.mysql.com/get/Downloads/Connector-J/mysql-connector-j-9.3.0.tar.gz
                                        Extract the archive                    =   tar -xzf mysql-connector-j-9.3.0.tar.gz

*Step 8  = (Create Java Source File):-  
                                       Create the Java source file  = touch StudentJDBC.java
                                       Open the file using Nano     = nano StudentJDBC.java
                                       Paste the Java program       ==
                                       
                                       
                                       
import java.sql.*;
public class StudentJDBC {

    static final String URL = "jdbc:mysql://localhost:3306/studentdb";
    static final String USER = "javauser";
    static final String PASSWORD = "password";

    public static void main(String args[]) {

        try {

            // Load Driver
            Class.forName("com.mysql.cj.jdbc.Driver");

            // Connect
            Connection con =
                    DriverManager.getConnection(URL, USER, PASSWORD);

            Statement stmt = con.createStatement();

            // Create Table
            String createTable =
                    "CREATE TABLE IF NOT EXISTS student (" +
                    "rollno INT PRIMARY KEY," +
                    "name VARCHAR(50)," +
                    "address VARCHAR(100))";

            stmt.execute(createTable);

            System.out.println("Table created.");

            // Insert Records
            stmt.executeUpdate(
                    "INSERT INTO student VALUES (1,'Ravi','Chennai')");

            stmt.executeUpdate(
                    "INSERT INTO student VALUES (2,'John','Delhi')");

            stmt.executeUpdate(
                    "INSERT INTO student VALUES (3,'Anita','Bangalore')");

            System.out.println("Records Inserted.");

            // Update
            stmt.executeUpdate(
                    "UPDATE student SET address='Mumbai' WHERE rollno=2");

            System.out.println("Record Updated.");

            // Delete
            stmt.executeUpdate(
                    "DELETE FROM student WHERE rollno=3");

            System.out.println("Record Deleted.");

            // Display

            ResultSet rs =
                    stmt.executeQuery("SELECT * FROM student");

            System.out.println("--------------------------------------");

            while(rs.next()) {

                System.out.println(
                        rs.getInt("rollno")+"  "
                        +rs.getString("name")+"  "
                        +rs.getString("address"));
            }

            System.out.println("--------------------------------------");

            rs.close();
            stmt.close();
            con.close();

        }

        catch(Exception e) {
            e.printStackTrace();
        }

    }
}
